cmake_minimum_required(VERSION 3.22)
project(vaind_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
add_subdirectory(../vendor/llama.cpp llama)
if(ANDROID)
 add_library(vaind_ai SHARED vaind_ai.cpp)
 target_link_libraries(vaind_ai PRIVATE llama android log)
 target_link_options(vaind_ai PRIVATE "-Wl,-z,max-page-size=16384")
else()
 add_executable(vaind-ai-check vaind_ai.cpp)
 target_link_libraries(vaind-ai-check PRIVATE llama)
endif()
